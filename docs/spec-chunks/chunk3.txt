### 7.2.4 BaseQuery Adapter Integration

#### 7.2.4.1 Rationale

BaseQuery (`g666st/BaseQuery`) generates SQLite databases from large breach compilations
such as COMB and Collection #1–#5. Its schema is well-known, consistently structured,
and used widely in the OSINT community for offline email → password lookup. Adding a
`BaseQueryAdapter` to Module 2-A's breach-DB abstraction layer gives FORGE native offline
access to multi-billion record credential sets without requiring custom ETL beyond an
initial import step.

> **OPSEC Note:** COMB and similar compilations are massive. Import must be performed
> during the Phase 0 pre-engagement window only. Do not import or query these datasets
> from engagement infrastructure. Use `data/` as the local cache directory and ensure
> it is excluded from any artefact exfiltration paths.

#### 7.2.4.2 BaseQuery Schema (Expected)

```sql
-- Expected tables in a BaseQuery-generated database
CREATE TABLE data (
    id       INTEGER PRIMARY KEY AUTOINCREMENT,
    email    TEXT NOT NULL,
    password TEXT
);
CREATE INDEX idx_email ON data(email);
```

`BaseQueryAdapter._validate_schema()` must confirm the presence of `data`, `email`, and
`password` columns before trusting the file. Raise `ValueError` with a clear message
if schema is unexpected.

#### 7.2.4.3 BaseQueryAdapter Class

```python
# forge/phase2/breach_db.py  (addition to existing BreachAdapter hierarchy)

class BaseQueryAdapter(BreachAdapter):
    """
    Reads a BaseQuery-format SQLite breach database.
    Schema: table 'data' with columns (id, email, password).
    """

    SOURCE_NAME = "basequery"

    def __init__(self, db_path: Path) -> None:
        self.db_path = db_path
        self._validate_schema()

    def _validate_schema(self) -> None:
        con = sqlite3.connect(self.db_path)
        try:
            cur = con.execute("SELECT name FROM sqlite_master WHERE type='table'")
            tables = {row[0] for row in cur.fetchall()}
            if "data" not in tables:
                raise ValueError(
                    f"BaseQuery schema validation failed: 'data' table not found in {self.db_path}"
                )
            cur = con.execute("PRAGMA table_info(data)")
            cols = {row[1] for row in cur.fetchall()}
            required = {"email", "password"}
            if not required.issubset(cols):
                missing = required - cols
                raise ValueError(
                    f"BaseQuery schema validation failed: missing columns {missing}"
                )
        finally:
            con.close()

    def records(self, emails: set[str]) -> Iterator[BreachRecord]:
        con = sqlite3.connect(self.db_path)
        con.row_factory = sqlite3.Row
        try:
            placeholders = ",".join("?" * len(emails))
            cur = con.execute(
                f"SELECT email, password FROM data WHERE email IN ({placeholders})",
                list(emails),
            )
            for row in cur:
                yield BreachRecord(
                    email=row["email"],
                    plaintext=row["password"] if not _looks_like_hash(row["password"]) else None,
                    hash_value=row["password"] if _looks_like_hash(row["password"]) else None,
                    source=self.SOURCE_NAME,
                )
        finally:
            con.close()
```

#### 7.2.4.4 Phase 0 BaseQuery Cache Build Script

```python
# scripts/build_basequery_cache.py
"""
Phase 0 pre-engagement utility.

Builds a BaseQuery-compatible SQLite database from raw COMB /
Breach Compilation text files. Input format: one credential per line,
delimited by ':' (email:password or email:hash).

Usage:
    python scripts/build_basequery_cache.py \
        --input /mnt/breach/COMB.txt \
        --output data/forge_comb.db
"""

from __future__ import annotations
import argparse
import sqlite3
from pathlib import Path

SCHEMA = """
CREATE TABLE IF NOT EXISTS data (
    id       INTEGER PRIMARY KEY AUTOINCREMENT,
    email    TEXT NOT NULL,
    password TEXT
);
"""
INDEX = "CREATE INDEX IF NOT EXISTS idx_email ON data(email);"


def build_cache(input_path: Path, output_path: Path, batch_size: int = 50_000) -> None:
    con = sqlite3.connect(output_path)
    con.execute("PRAGMA journal_mode=WAL")
    con.execute("PRAGMA synchronous=NORMAL")
    con.execute(SCHEMA)
    con.commit()

    total = 0
    batch: list[tuple[str, str]] = []

    with open(input_path, "r", errors="replace") as fh:
        for line in fh:
            line = line.strip()
            if ":" not in line:
                continue
            email, _, password = line.partition(":")
            if not email or "@" not in email:
                continue
            batch.append((email.lower(), password))
            if len(batch) >= batch_size:
                con.executemany(
                    "INSERT INTO data (email, password) VALUES (?, ?)", batch
                )
                con.commit()
                total += len(batch)
                batch.clear()
                print(f"\r  Imported {total:,} records...", end="", flush=True)

    if batch:
        con.executemany("INSERT INTO data (email, password) VALUES (?, ?)", batch)
        con.commit()
        total += len(batch)

    print(f"\n  Building email index on {total:,} records...")
    con.execute(INDEX)
    con.commit()
    con.close()
    print(f"  Done. Database written to: {output_path}")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Build FORGE BaseQuery-compatible breach cache")
    parser.add_argument("--input",      type=Path, required=True)
    parser.add_argument("--output",     type=Path, default=Path("data/forge_comb.db"))
    parser.add_argument("--batch-size", type=int,  default=50_000)
    args = parser.parse_args()
    build_cache(args.input, args.output, args.batch_size)
```

### 7.2.5 pwndb Legacy Dataset Support

#### 7.2.5.1 Rationale

The pwndb Tor hidden service (`pwndb2am4tzkvold.onion`) is operationally unreliable and
its response times over Tor make live querying impractical at engagement scale. Many OSINT
practitioners have downloaded local snapshots of the pwndb dataset in raw line-separated
format. FORGE supports importing these dumps into the local breach cache for offline
querying without any network dependency.

> **OPSEC Note:** Do not route live queries through Tor without operator-controlled exit
> infrastructure. Session correlation via Tor timing attacks is a documented OPSEC risk.
> The offline import path is strongly preferred.

#### 7.2.5.2 Phase 0 Import Script

```python
# scripts/import_pwndb_dump.py
"""
Phase 0 pre-engagement utility.

Imports a locally-held pwndb dataset dump into FORGE's local breach cache.
The pwndb dump format is one credential per line, delimited by ':'

    email:password_or_hash

This script normalises the data and inserts it into an SQLite file
compatible with TextBreachAdapter (or BaseQueryAdapter if --target-format
basequery is specified).
"""
from __future__ import annotations
import argparse
import sqlite3
from pathlib import Path

SCHEMA = """
CREATE TABLE IF NOT EXISTS data (
    id       INTEGER PRIMARY KEY AUTOINCREMENT,
    email    TEXT NOT NULL,
    password TEXT,
    source   TEXT NOT NULL DEFAULT 'pwndb',
    UNIQUE(email, password, source)
);
"""


def import_dump(
    input_path:   Path,
    output_path:  Path,
    breach_name:  str = "pwndb",
    batch_size:   int = 50_000,
) -> int:
    con = sqlite3.connect(output_path)
    con.execute("PRAGMA journal_mode=WAL")
    con.execute("PRAGMA synchronous=NORMAL")
    con.execute(SCHEMA)
    con.commit()

    total = 0
    batch: list[tuple[str, str, str]] = []

    with open(input_path, "r", errors="replace") as fh:
        for line in fh:
            line = line.strip()
            if ":" not in line:
                continue
            email, _, password = line.partition(":")
            if not email or "@" not in email:
                continue
            batch.append((email.lower(), password, breach_name))
            if len(batch) >= batch_size:
                con.executemany(
                    "INSERT OR IGNORE INTO data (email, password, source) VALUES (?, ?, ?)",
                    batch,
                )
                con.commit()
                total += len(batch)
                batch.clear()

    if batch:
        con.executemany(
            "INSERT OR IGNORE INTO data (email, password, source) VALUES (?, ?, ?)",
            batch,
        )
        con.commit()
        total += len(batch)

    con.close()
    return total


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Import pwndb dump into FORGE breach cache")
    parser.add_argument("--input",        type=Path, required=True)
    parser.add_argument("--output",       type=Path, default=Path("data/pwndb_cache.db"))
    parser.add_argument("--breach-name",  type=str,  default="pwndb")
    parser.add_argument("--batch-size",   type=int,  default=50_000)
    args = parser.parse_args()
    n = import_dump(args.input, args.output, args.breach_name, args.batch_size)
    print(f"Imported {n:,} records → {args.output}")
```

### 7.3.5 HashBuster Pre-Spray Hash Cracking

#### 7.3.5.1 Rationale

Breach records harvested by Module 2-A frequently contain hashed rather than plaintext
passwords. Module 2-B cannot validate a hash against a live authentication target without
first resolving it. The existing implementation skips hash-bearing credentials entirely,
leaving a material gap in spray coverage. A HashBuster integration resolves this gap by
querying multiple online hash lookup databases (Google cache, MD5Online, Nitrxgen,
md5decrypt) before the spray queue is built, elevating cracked hashes to plaintext and
making them sprayable.

> **Rate-limiting consideration:** HashBuster queries public, unauthenticated endpoints.
> These services impose implicit rate limits enforced by HTTP 429 responses and soft
> blocks. The implementation must honour exponential backoff per endpoint and skip
> permanently to the next endpoint on a hard block rather than retrying indefinitely.
> Total cracking time should be bounded by `--crack-timeout` (default: 30 s per hash).

> **OPSEC Note:** HashBuster resolves hashes against external online services. These
> queries leave a fingerprint correlating specific hash values to your operator IP. Use
> a VPN or proxy for hash cracking that is separate from the engagement infrastructure.
> Alternatively, use `--crack-offline` to restrict cracking to `hashcat` / local
> wordlists only.

#### 7.3.5.2 HashCracker Utility

```python
# forge/utils/hash_cracker.py
"""
Offline-first hash cracking utility used by Module 2-B as a pre-spray gate.

Online mode: queries md5decrypt, Nitrxgen (sequential, with exponential backoff).
Offline mode: invokes hashcat subprocess with operator-supplied wordlist.

All cracked plaintexts are age-encrypted before DB write — never stored in plaintext.
Bcrypt ($2a$, $2b$, $2y$) hashes are skipped unconditionally.
"""
from __future__ import annotations

import hashlib
import logging
import subprocess
import time
from enum import Enum
from typing import Optional

import httpx

logger = logging.getLogger(__name__)


class HashType(str, Enum):
    MD5    = "md5"
    NTLM   = "ntlm"
    SHA1   = "sha1"
    SHA256 = "sha256"
    BCRYPT = "bcrypt"
    UNKNOWN = "unknown"


ENDPOINTS = [
    "https://md5decrypt.net/Api/api.php?hash={hash}&hash_type={type}&email=api@forge&code=FREE_CODE",
    "https://www.nitrxgen.net/md5db/{hash}",
]


def _infer_hash_type(value: str) -> HashType:
    if value.startswith(("$2a$", "$2b$", "$2y$")):
        return HashType.BCRYPT
    length = len(value)
    if length == 32:
        return HashType.MD5
    if length == 40:
        return HashType.SHA1
    if length == 64:
        return HashType.SHA256
    if length == 32 and value == value.upper():
        return HashType.NTLM
    return HashType.UNKNOWN


def crack_hash(
    hash_value: str,
    proxy:      Optional[str] = None,
    timeout:    int = 30,
    base_delay: float = 2.0,
) -> Optional[str]:
    """
    Attempt to resolve a single hash to plaintext.
    Returns plaintext string on success, None on failure.
    Skips bcrypt unconditionally.
    """
    ht = _infer_hash_type(hash_value)
    if ht == HashType.BCRYPT:
        logger.debug("Skipping bcrypt hash (unsupported for online cracking).")
        return None

    transport = httpx.HTTPTransport(proxy=proxy) if proxy else None

    for endpoint_template in ENDPOINTS:
        url = endpoint_template.format(hash=hash_value, type=ht.value)
        attempt = 0
        while attempt < 3:
            try:
                with httpx.Client(transport=transport, timeout=timeout) as client:
                    resp = client.get(url)
                if resp.status_code == 200 and resp.text.strip():
                    plaintext = resp.text.strip()
                    # Basic sanity: verify hash matches
                    if _verify(hash_value, ht, plaintext):
                        return plaintext
                    break  # bad result, next endpoint
                if resp.status_code == 429:
                    wait = base_delay * (2 ** attempt)
                    logger.warning("Rate limited by %s — waiting %.1fs", url, wait)
                    time.sleep(wait)
                    attempt += 1
                    continue
                break  # non-200 non-429: skip endpoint
            except httpx.RequestError as exc:
                logger.debug("HashCracker request error: %s", exc)
                break

    return None


def _verify(hash_value: str, ht: HashType, plaintext: str) -> bool:
    encoded = plaintext.encode()
    if ht == HashType.MD5:
        return hashlib.md5(encoded).hexdigest() == hash_value.lower()
    if ht == HashType.SHA1:
        return hashlib.sha1(encoded).hexdigest() == hash_value.lower()
    if ht == HashType.SHA256:
        return hashlib.sha256(encoded).hexdigest() == hash_value.lower()
    return True  # NTLM: skip verification (requires NT-hash computation)


def _crack_with_hashcat(
    hash_value: str,
    wordlist:   str,
    ht:         HashType,
) -> Optional[str]:
    """Offline fallback using hashcat subprocess."""
    mode_map = {
        HashType.MD5:    "0",
        HashType.NTLM:   "1000",
        HashType.SHA1:   "100",
        HashType.SHA256: "1400",
    }
    mode = mode_map.get(ht)
    if not mode:
        return None
    try:
        result = subprocess.run(
            ["hashcat", "-m", mode, hash_value, wordlist, "--quiet", "--potfile-disable"],
            capture_output=True, text=True, timeout=120,
        )
        for line in result.stdout.splitlines():
            if ":" in line:
                _, _, cracked = line.rpartition(":")
                return cracked.strip() or None
    except (FileNotFoundError, subprocess.TimeoutExpired):
        logger.debug("hashcat not available or timed out.")
    return None


def crack_batch(
    hashes:     list[str],
    proxy:      Optional[str] = None,
    offline:    bool = False,
    wordlist:   Optional[str] = None,
    timeout:    int = 30,
) -> dict[str, Optional[str]]:
    """Crack a batch of hashes. Returns {hash_value: plaintext_or_None}."""
    results: dict[str, Optional[str]] = {}
    for h in hashes:
        if offline and wordlist:
            ht = _infer_hash_type(h)
            results[h] = _crack_with_hashcat(h, wordlist, ht)
        else:
            results[h] = crack_hash(h, proxy=proxy, timeout=timeout)
    return results
```

For air-gapped environments, use `--crack-offline --wordlist /path/rockyou.txt` to

### 7.7.5 LeakLooker Paste Monitoring Integration

#### 7.7.5.1 Rationale

Module 2-F's emailrep.io lookup is a point-in-time enrichment: it captures breach

> **OPSEC Note:** Paste site polling is entirely passive — no target systems are queried.

#### 7.7.5.2 Paste Monitor Implementation

```python
# forge/phase2/paste_monitor.py
"""
Optional real-time paste monitoring for Module 2-F.
Polls Pastebin (and optionally other paste sites) for operator-defined keywords.
Stores hits in paste_alerts table. Minimum poll interval: 30 seconds.
"""
from __future__ import annotations

import sqlite3
import time
import logging
from dataclasses import dataclass
from pathlib import Path
from typing import Optional

import httpx

logger = logging.getLogger(__name__)

PASTEBIN_SCRAPE_URL = "https://scrape.pastebin.com/api_scraping.php?limit=100"
PASTEBIN_RAW_URL    = "https://scrape.pastebin.com/api_scrape_item.php?i={key}"
PASTEBIN_RSS_URL    = "https://pastebin.com/rss/"
MIN_POLL_INTERVAL   = 30   # seconds


@dataclass
class PasteAlert:
    keyword:   str
    paste_url: str
    snippet:   str    # truncated to 512 chars


class PasteMonitor:
    def __init__(
        self,
        db_path:       Path,
        engagement_id: int,
        keywords:      list[str],
        poll_interval: int = 60,
        pro_account:   bool = False,
        proxy:         Optional[str] = None,
    ) -> None:
        self.db_path       = db_path
        self.engagement_id = engagement_id
        self.keywords      = [kw.lower() for kw in keywords]
        self.poll_interval = max(poll_interval, MIN_POLL_INTERVAL)
        self.pro_account   = pro_account
        self.proxy         = proxy

    def run(self) -> None:
        """Block and poll until interrupted."""
        logger.info("Paste monitor started. Poll interval: %ds", self.poll_interval)
        while True:
            try:
                pastes = self._fetch_recent_pastes()
                for paste in pastes:
                    content = self._fetch_paste_content(paste["key"])
                    if content:
                        for kw in self.keywords:
                            if kw in content.lower():
                                snippet = content[:512]
                                self._save_alert(PasteAlert(
                                    keyword=kw,
                                    paste_url=f"https://pastebin.com/{paste['key']}",
                                    snippet=snippet,
                                ))
            except Exception as exc:
                logger.warning("Paste monitor cycle error: %s", exc)
            time.sleep(self.poll_interval)

    def _fetch_recent_pastes(self) -> list[dict]:
        if self.pro_account:
            resp = httpx.get(PASTEBIN_SCRAPE_URL, timeout=10)
            if resp.status_code == 200:
                return resp.json()
        # RSS fallback
        resp = httpx.get(PASTEBIN_RSS_URL, timeout=10)
        return self._parse_rss(resp.text)

    def _fetch_paste_content(self, key: str) -> Optional[str]:
        try:
            resp = httpx.get(PASTEBIN_RAW_URL.format(key=key), timeout=10)
            return resp.text if resp.status_code == 200 else None
        except httpx.RequestError:
            return None

    def _parse_rss(self, xml: str) -> list[dict]:
        import xml.etree.ElementTree as ET
        items = []
        try:
            root = ET.fromstring(xml)
            for item in root.findall(".//item"):
                link = item.findtext("link", "")
                key  = link.rsplit("/", 1)[-1] if link else ""
                if key:
                    items.append({"key": key})
        except ET.ParseError:
            pass
        return items

    def _save_alert(self, alert: PasteAlert) -> None:
        con = sqlite3.connect(self.db_path)
        con.execute("""
            INSERT OR IGNORE INTO paste_alerts
                (engagement_id, keyword, paste_url, snippet)
            VALUES (?, ?, ?, ?)
        """, (self.engagement_id, alert.keyword, alert.paste_url, alert.snippet))
        con.commit()
        con.close()
```

Alerts are stored in the `paste_alerts` table and automatically included in the
"New Paste Alerts Since Last Report" section of the Phase 6 report. A Pastebin Pro
account (`--pastebin-pro`) provides higher fidelity via the scraping API; free-tier
accounts fall back to RSS. Minimum polling interval is 30 seconds. Use
`--monitor-proxy` to route monitoring traffic independently from engagement
infrastructure.

## §7.11 — Module 2-J: KeyScanner — Exposed API Key Discovery & Validation (NEW)

### 7.11.1 Justification

Module 2-I (Scavenger) performs domain-keyed secret discovery: given a target domain it
finds any secret that co-appears with that domain in public code. Module 2-J inverts the
search axis: it is **pattern-keyed**. It queries for known API key regex patterns directly
(e.g., `AKIA[0-9A-Z]{16}` for AWS) filtered by domain or organisation name, then
**actively validates each discovered key** against the relevant provider API to confirm
whether it is still live.

A confirmed `ACTIVE` key is a higher-severity, more immediately actionable finding than
a generic secret found by Scavenger. Active AWS keys feed Phase 4 cloud exploit scoring
and Phase 5 lateral movement via cloud provider APIs. Active GitHub PATs may expose
private repositories, CI secrets, or code signing keys with no additional exploitation
steps required.

> **Positioning relative to Module 2-I:**
> - Module 2-I: `domain → find secrets mentioning that domain`
> - **Module 2-J: `key pattern → find matching keys attributed to domain → confirm liveness`**

> **OPSEC:** GitHub code search queries are logged by GitHub and attributed to the PAT
> used. Use a purpose-built, non-attributable throwaway account — **never an operator's
> personal account**. All validation calls must be routed through `--validation-proxy`.
> The CLI exits code `1` if `--validation-proxy` is absent and `--no-validate` is not set.

---

```python
Module 2-J — Exposed API Key Discovery & Active Validation.

Performs pattern-keyed searches of public code repositories for
API keys attributed to a target domain, then validates each discovered
key against the relevant provider API.

import sqlite3

# ── Enums & data classes ───────────────────────────────────────────────────

class ValidationState(str, Enum):
    ACTIVE      = "ACTIVE"
    REVOKED     = "REVOKED"
    UNCONFIRMED = "UNCONFIRMED"
    ERROR       = "ERROR"


@dataclass
class KeyPattern:
    name:              str
    service:           str
    regex:             re.Pattern
    confidence:        str
    validation_method: Optional[str]
    group:             int = 0


@dataclass
class ValidationResult:
    state:  ValidationState
    detail: Optional[str] = None


# ── Pattern loading ────────────────────────────────────────────────────────

def load_key_patterns(path: Path = PATTERN_FILE) -> list[KeyPattern]:
    with open(path) as f:
        data = json.load(f)
    return [
        KeyPattern(
            name=p["name"],
            service=p["service"],
            regex=re.compile(p["regex"], re.MULTILINE),
            confidence=p.get("confidence", "medium"),
            validation_method=p.get("validation_method"),
            group=p.get("group", 0),
        )
        for p in data.get("patterns", [])
    ]


# ── Redaction ──────────────────────────────────────────────────────────────

def _redact(value: str) -> str:
    """Return first4...last4; '****' for keys ≤ 8 chars."""
    if len(value) <= 8:
        return "****"
    return f"{value[:4]}...{value[-4:]}"


# ── Validators ─────────────────────────────────────────────────────────────

class BaseKeyValidator(ABC):
    """Abstract base. All validators must route calls through proxy if provided."""

    @abstractmethod
    def validate(
        self,
        key:   str,
        proxy: Optional[str] = None,
        **kwargs,
    ) -> ValidationResult: ...


class GithubPatValidator(BaseKeyValidator):
    """GET /user with Bearer token: 200 → ACTIVE, 401 → REVOKED."""
    API_URL = "https://api.github.com/user"

    def validate(self, key: str, proxy: Optional[str] = None, **kwargs) -> ValidationResult:
        transport = httpx.HTTPTransport(proxy=proxy) if proxy else None
        with httpx.Client(transport=transport, timeout=10) as client:
            resp = client.get(
                self.API_URL,
                headers={"Authorization": f"Bearer {key}",
                         "Accept": "application/vnd.github+json"},
            )
        if resp.status_code == 200:
            login = resp.json().get("login", "unknown")
            return ValidationResult(state=ValidationState.ACTIVE, detail=f"GitHub login: {login}")
        if resp.status_code == 401:
            return ValidationResult(state=ValidationState.REVOKED)
        return ValidationResult(state=ValidationState.ERROR, detail=f"HTTP {resp.status_code}")


class StripeKeyValidator(BaseKeyValidator):
    """GET /v1/balance: 200 → ACTIVE, 401 → REVOKED."""
    API_URL = "https://api.stripe.com/v1/balance"

    def validate(self, key: str, proxy: Optional[str] = None, **kwargs) -> ValidationResult:
        transport = httpx.HTTPTransport(proxy=proxy) if proxy else None
        with httpx.Client(transport=transport, timeout=10) as client:
            resp = client.get(self.API_URL, auth=(key, ""))
        if resp.status_code == 200:
            return ValidationResult(state=ValidationState.ACTIVE, detail="Stripe balance endpoint OK")
        if resp.status_code == 401:
            return ValidationResult(state=ValidationState.REVOKED)
        return ValidationResult(state=ValidationState.ERROR, detail=f"HTTP {resp.status_code}")


class AwsKeyValidator(BaseKeyValidator):
    """
    STS GetCallerIdentity via SigV4-signed request (boto3-free, curl_cffi).
    Requires both access key ID and secret key to be co-located in source file.
    Returns UNCONFIRMED if secret is absent.
    """

    def validate(
        self,
        key:    str,
        proxy:  Optional[str] = None,
        secret: Optional[str] = None,
        **kwargs,
    ) -> ValidationResult:
        if not secret:
            return ValidationResult(
                state=ValidationState.UNCONFIRMED,
                detail="AWS secret key not co-located with access key ID in source file",
            )
        # SigV4-signed POST to https://sts.amazonaws.com/ via curl_cffi
        # Returns ACTIVE + AccountId on success, REVOKED on AuthFailure
        raise NotImplementedError("AwsKeyValidator SigV4 implementation required — see task 2J.7")


# Validator registry — keyed on validation_method string from pattern file
VALIDATORS: dict[str, BaseKeyValidator] = {
    "GithubPatValidator": GithubPatValidator(),
    "StripeKeyValidator": StripeKeyValidator(),
    "AwsKeyValidator":    AwsKeyValidator(),
    # SendgridKeyValidator and TwilioKeyValidator: implement analogously (task 2J.8)
}


# ── GitHub search backend ──────────────────────────────────────────────────

def _github_keyscan(
    domain:   str,
    patterns: list[KeyPattern],
    token:    Optional[str],
    client:   httpx.Client,
    delay:    float,
) -> Iterator[tuple[KeyPattern, str, dict]]:
    """
    For each high-confidence pattern, queries GitHub Code Search for files
    containing the pattern AND the target domain. Yields (pattern, raw_key, result_dict).
    """
    headers = {"Accept": "application/vnd.github+json"}
    if token:
        headers["Authorization"] = f"Bearer {token}"

    for pat in patterns:
        if pat.confidence not in ("high",):
            continue

        query = f'"{domain}" "{pat.regex.pattern[:20]}"'
        page  = 1

        while True:
            resp = client.get(
                GITHUB_SEARCH_URL,
                params={"q": query, "per_page": 30, "page": page},
                headers=headers,
                timeout=15,
            )

            if resp.status_code == 429 or resp.headers.get("X-RateLimit-Remaining") == "0":
                reset = int(resp.headers.get("X-RateLimit-Reset", time.time() + 60))
                wait  = max(reset - time.time(), 1)
                logger.warning("GitHub rate limit — waiting %.0fs", wait)
                time.sleep(wait)
                continue

            if resp.status_code != 200:
                logger.warning("GitHub search returned %d for pattern %s", resp.status_code, pat.name)
                break

            items = resp.json().get("items", [])
            if not items:
                break

            for item in items:
                raw_url = item.get("html_url", "").replace(
                    "github.com", "raw.githubusercontent.com"
                ).replace("/blob/", "/")
                try:
                    raw_resp = client.get(raw_url, headers=headers, timeout=15)
                    content  = raw_resp.text if raw_resp.status_code == 200 else ""
                except httpx.RequestError:
                    content = ""

                for match in pat.regex.finditer(content):
                    raw_key = match.group(pat.group)
                    yield pat, raw_key, {
                        "url":       item["html_url"],
                        "repo_name": item.get("repository", {}).get("full_name"),
                        "backend":   "github",
                        "_raw_key":  raw_key,
                    }

            if len(items) < 30:
                break
            page += 1
            time.sleep(delay)


# ── Storage ────────────────────────────────────────────────────────────────

def _store_finding(
    con:           sqlite3.Connection,
    engagement_id: int,
    domain:        str,
    pat:           KeyPattern,
    result:        dict,
    validation:    ValidationResult,
    age_pubkey:    str,
) -> bool:
    """Inserts finding; returns True if new row was inserted."""
    key_enc = encrypt_string(result.get("_raw_key", ""), pubkey=age_pubkey)
    cur = con.execute(
        """INSERT OR IGNORE INTO key_scanner_findings
           (engagement_id, domain, service, pattern_name, source_backend,
            source_url, repo_name, key_redacted, key_enc,
            validation_state, validation_detail, validated_at)
           VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)""",
        (
            engagement_id, domain, pat.service, pat.name,
            result.get("backend", "github"),
            result["url"], result.get("repo_name"),
            _redact(result.get("_raw_key", "")),
            key_enc,
            validation.state.value,
            validation.detail,
        ),
    )
    con.commit()
    return cur.rowcount > 0


# ── Orchestrator ───────────────────────────────────────────────────────────

def run_keyscan(
    engagement_id:    int,
    db_path:          Path,
    domain:           str,
    github_token:     Optional[str],
    gitlab_token:     Optional[str],
    validation_proxy: Optional[str],
    no_validate:      bool,
    delay:            float,
    age_pubkey:       str,
    dry_run:          bool = False,
) -> int:
    """
    Main entry point for Module 2-J. Returns total new findings stored.

    Proxy enforcement: if not no_validate and not validation_proxy → sys.exit(1).
    """
    assert_in_scope(domain)

    if not no_validate and not validation_proxy:
        import click
        click.echo(
            "[OPSEC ERROR] --validation-proxy is required for active key validation.\n"
            "Use --no-validate to skip validation, or supply --validation-proxy socks5://..."
        )
        sys.exit(1)

    if not no_validate:
        confirmed = questionary.confirm(
            f"[Module 2-J] About to validate API keys against external provider APIs.\n"
            f"  Domain:           {domain}\n"
            f"  Validation proxy: {validation_proxy}\n"
            f"  These calls are logged by the provider and may trigger alerts.\n"
            f"Continue?"
        ).ask()
        if not confirmed:
            raise RuntimeError("Operator cancelled key validation.")

    patterns = load_key_patterns()
    con      = sqlite3.connect(db_path)
    total    = 0

    with httpx.Client(timeout=15) as client:
        for pat, raw_key, result in _github_keyscan(domain, patterns, github_token, client, delay):
            if dry_run:
                logger.info("[DRY RUN] Found %s key at %s", pat.name, result["url"])
                continue

            # Dedup against scavenger_findings
            existing = con.execute(
                "SELECT 1 FROM scavenger_findings WHERE source_url=? AND pattern_name=?",
                (result["url"], pat.name),
            ).fetchone()
            if existing:
                logger.debug("Skipping %s — already in scavenger_findings", result["url"])
                continue

            if no_validate or not pat.validation_method:
                validation = ValidationResult(state=ValidationState.UNCONFIRMED)
            else:
                validator  = VALIDATORS.get(pat.validation_method)
                validation = validator.validate(raw_key, proxy=validation_proxy) if validator \
                             else ValidationResult(state=ValidationState.UNCONFIRMED,
                                                   detail="[NO VALIDATOR]")

            if _store_finding(con, engagement_id, domain, pat, result, validation, age_pubkey):
                total += 1
                logger.info(
                    "[2-J] %s | %s | %s | %s",
                    pat.service.upper(), pat.name, _redact(raw_key), validation.state.value,
                )

    con.close()
    return total
```
